DROP TABLE audit_log CASCADE CONSTRAINTS;
DROP TABLE valeur_ligne CASCADE CONSTRAINTS;
DROP TABLE ligne_facture CASCADE CONSTRAINTS;
DROP TABLE facture CASCADE CONSTRAINTS;
DROP TABLE colonne CASCADE CONSTRAINTS;
DROP TABLE item CASCADE CONSTRAINTS;
DROP TABLE concession CASCADE CONSTRAINTS;

-- Supprimer les séquences
DROP SEQUENCE concession_seq;
DROP SEQUENCE item_seq;
DROP SEQUENCE colonne_seq;
DROP SEQUENCE facture_seq;
DROP SEQUENCE ligne_facture_seq;
DROP SEQUENCE valeur_ligne_seq;
DROP SEQUENCE audit_log_seq;

-- ============================================================
-- TABLES
-- ============================================================

CREATE TABLE concession (
    id_concession   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom             VARCHAR2(200) NOT NULL,
    nom_normalise   VARCHAR2(200) NOT NULL,
    date_creation   TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT uniq_concession UNIQUE (nom_normalise)
);

CREATE TABLE item (
    id_item             NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_concession       NUMBER NOT NULL,
    libelle_canonique   VARCHAR2(300) NOT NULL,
    libelle_recherche   VARCHAR2(300) NOT NULL,
    statut              VARCHAR2(20) DEFAULT 'actif',
    date_creation       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    date_derniere_util  TIMESTAMP NULL,
    fusionne_avec       NUMBER NULL,
    CONSTRAINT fk_item_concession FOREIGN KEY (id_concession) REFERENCES concession(id_concession),
    CONSTRAINT uniq_item UNIQUE (id_concession, libelle_canonique)
);

CREATE TABLE colonne (
    id_colonne          NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_concession       NUMBER NOT NULL,
    libelle_canonique   VARCHAR2(200) NOT NULL,
    libelle_recherche   VARCHAR2(200) NOT NULL,
    date_creation       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_colonne_concession FOREIGN KEY (id_concession) REFERENCES concession(id_concession),
    CONSTRAINT uniq_colonne UNIQUE (id_concession, libelle_canonique)
);

CREATE TABLE facture (
    id_facture      NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_concession   NUMBER NOT NULL,
    numero_facture  VARCHAR2(100),
    date_facture    TIMESTAMP,
    date_extraction TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total_montant   FLOAT,
    total_quantite  FLOAT,
    devise          VARCHAR2(10) DEFAULT 'EUR',
    fichier_source  VARCHAR2(500),
    hash_contenu    VARCHAR2(200) UNIQUE,
    statut          VARCHAR2(20) DEFAULT 'traite',
    fournisseur     VARCHAR2(300),
    confiance_ocr   FLOAT,
    CONSTRAINT fk_facture_concession FOREIGN KEY (id_concession) REFERENCES concession(id_concession)
);

CREATE TABLE ligne_facture (
    id_ligne    NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_facture  NUMBER NOT NULL,
    id_item     NUMBER NOT NULL,
    confiance   FLOAT DEFAULT 1.0,
    auto_match  VARCHAR2(1) DEFAULT '1',
    CONSTRAINT fk_ligne_facture FOREIGN KEY (id_facture) REFERENCES facture(id_facture),
    CONSTRAINT fk_ligne_item FOREIGN KEY (id_item) REFERENCES item(id_item)
);

CREATE TABLE valeur_ligne (
    id_valeur       NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_ligne        NUMBER NOT NULL,
    id_colonne      NUMBER NOT NULL,
    valeur_brute    VARCHAR2(100) NOT NULL,
    valeur_numerique FLOAT NULL,
    CONSTRAINT fk_valeur_ligne FOREIGN KEY (id_ligne) REFERENCES ligne_facture(id_ligne),
    CONSTRAINT fk_valeur_colonne FOREIGN KEY (id_colonne) REFERENCES colonne(id_colonne),
    CONSTRAINT uniq_valeur UNIQUE (id_ligne, id_colonne)
);

CREATE TABLE audit_log (
    id_audit        NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    table_name      VARCHAR2(50) NOT NULL,
    action          VARCHAR2(10) NOT NULL,
    record_id       NUMBER,
    old_value       CLOB,
    new_value       CLOB,
    modified_by     VARCHAR2(100) DEFAULT 'system',
    modified_at     TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE fournisseurs (
    id_fournisseur   NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nom              VARCHAR2(255) NOT NULL UNIQUE,
    nom_normalise    VARCHAR2(255) NOT NULL,
    id_concession    NUMBER,
    created_at       TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_fournisseur_concession FOREIGN KEY (id_concession) REFERENCES concession(id_concession)
);

-- Index
CREATE INDEX idx_item_concession ON item(id_concession);
CREATE INDEX idx_item_recherche ON item(id_concession, libelle_recherche);
CREATE INDEX idx_colonne_concession ON colonne(id_concession);
CREATE INDEX idx_facture_concession ON facture(id_concession);
CREATE INDEX idx_facture_hash ON facture(hash_contenu);
CREATE INDEX idx_ligne_facture_fk ON ligne_facture(id_facture);
CREATE INDEX idx_ligne_item_fk ON ligne_facture(id_item);
CREATE INDEX idx_valeur_ligne_fk ON valeur_ligne(id_ligne);

-- ═══════════════════════════════════════════════════════════
-- 1. Créer la table section_facture (sans dépendances)
-- ═══════════════════════════════════════════════════════════
CREATE TABLE section_facture (
    id_section      NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_facture      NUMBER NOT NULL,
    section_index   NUMBER NOT NULL,
    titre           VARCHAR2(200) NOT NULL,
    CONSTRAINT fk_section_facture FOREIGN KEY (id_facture)
        REFERENCES facture(id_facture) ON DELETE CASCADE,
    CONSTRAINT uniq_section_order UNIQUE (id_facture, section_index)
);

CREATE INDEX idx_section_facture ON section_facture(id_facture);

-- ═══════════════════════════════════════════════════════════
-- 2. Créer la table section_colonne
-- ═══════════════════════════════════════════════════════════
CREATE TABLE section_colonne (
    id_section  NUMBER NOT NULL,
    id_colonne  NUMBER NOT NULL,
    ordre       NUMBER NOT NULL,
    CONSTRAINT pk_section_colonne PRIMARY KEY (id_section, id_colonne),
    CONSTRAINT fk_sectcol_section FOREIGN KEY (id_section)
        REFERENCES section_facture(id_section) ON DELETE CASCADE,
    CONSTRAINT fk_sectcol_colonne FOREIGN KEY (id_colonne)
        REFERENCES colonne(id_colonne)
);

CREATE INDEX idx_sectcol_section ON section_colonne(id_section);
CREATE INDEX idx_sectcol_colonne ON section_colonne(id_colonne);

-- ═══════════════════════════════════════════════════════════
-- 3. Ajouter id_section à ligne_facture (NULLABLE d'abord)
-- ═══════════════════════════════════════════════════════════
ALTER TABLE ligne_facture ADD (id_section NUMBER NULL);

-- ═══════════════════════════════════════════════════════════
-- 4. Créer une section par défaut pour chaque facture existante
-- ═══════════════════════════════════════════════════════════
INSERT INTO section_facture (id_facture, section_index, titre)
SELECT id_facture, 1, 'Section 1'
FROM facture
WHERE NOT EXISTS (
    SELECT 1 FROM section_facture WHERE section_facture.id_facture = facture.id_facture
);

-- ═══════════════════════════════════════════════════════════
-- 5. Mettre à jour les lignes existantes avec la section par défaut
-- ═══════════════════════════════════════════════════════════
UPDATE ligne_facture lf
SET lf.id_section = (
    SELECT sf.id_section
    FROM section_facture sf
    WHERE sf.id_facture = lf.id_facture
      AND sf.section_index = 1
)
WHERE lf.id_section IS NULL;

-- ═══════════════════════════════════════════════════════════
-- 6. Maintenant on peut rendre la colonne NOT NULL
-- ═══════════════════════════════════════════════════════════
ALTER TABLE ligne_facture MODIFY (id_section NUMBER NOT NULL);

-- ═══════════════════════════════════════════════════════════
-- 7. Ajouter la contrainte de clé étrangère
-- ═══════════════════════════════════════════════════════════
ALTER TABLE ligne_facture ADD CONSTRAINT fk_ligne_section
    FOREIGN KEY (id_section) REFERENCES section_facture(id_section);

CREATE INDEX idx_ligne_section_fk ON ligne_facture(id_section);

-- ═══════════════════════════════════════════════════════════
-- 8. Supprimer l'ancienne colonne section_index
-- ═══════════════════════════════════════════════════════════
ALTER TABLE ligne_facture DROP COLUMN section_index;

-- ═══════════════════════════════════════════════════════════
-- 9. Agrandir valeur_brute
-- ═══════════════════════════════════════════════════════════
ALTER TABLE valeur_ligne MODIFY (valeur_brute VARCHAR2(500));
ALTER TABLE facture ADD (extra_metadata VARCHAR2(4000))
